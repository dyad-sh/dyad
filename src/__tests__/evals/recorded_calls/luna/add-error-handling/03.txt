export async function patchResource<T>(
  path: string,
  body: unknown,
): Promise<T> {
  try {
    const data = await serviceRequest<T>(path, { method: "PATCH", body });
    return data;
  } catch (err) {
    logger.error(`PATCH ${path} failed`, err);
    throw err;
  }
}
